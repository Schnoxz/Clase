Un procedimiento almacenado es un bloque de SQL guardado en el servidor que se invoca con CALL. La primera ejecución se compila y guarda en caché para las siguientes llamadas.
-------------------------------------------------------------------------
DELIMITER $$
CREATE PROCEDURE GetClientes()
BEGIN
    SELECT nombre, ciudad, pais
    FROM Clientes
    ORDER BY nombre;
END$$
DELIMITER ;

-- Invocación
CALL GetClientes();

Se usa DELIMITER $$ para que el cliente MySQL no interprete los puntos y coma internos como fin de sentencia.

Variables y parámetros IN
-------------------------------------------------------------------------
Las variables locales se declaran con DECLARE y se asignan con SET o SELECT INTO. Los parámetros IN permiten pasar valores al procedimiento.

DELIMITER $$
CREATE PROCEDURE GetOficinasPais(IN pPais VARCHAR(100))
BEGIN
    DECLARE totalOficinas INT DEFAULT 0;

    SELECT COUNT(*) INTO totalOficinas
    FROM Oficinas WHERE pais = pPais;

    SELECT * FROM Oficinas WHERE pais = pPais;
    SELECT totalOficinas AS 'Total encontradas';
END$$
DELIMITER ;

CALL GetOficinasPais('España');

Funciones almacenadas
-------------------------------------------------------------------------
A diferencia de un procedimiento, una función devuelve un único valor y puede usarse dentro de una sentencia SELECT.

DELIMITER $$
CREATE FUNCTION NivelCliente(credito DECIMAL(10,2))
RETURNS VARCHAR(20)
BEGIN
    DECLARE nivel VARCHAR(20);
    IF credito > 50000 THEN
        SET nivel = 'PLATINUM';
    ELSEIF credito >= 10000 THEN
        SET nivel = 'ORO';
    ELSE
        SET nivel = 'PLATA';
    END IF;
    RETURN nivel;
END$$
DELIMITER ;

-- Uso en una consulta
SELECT nombre, NivelCliente(limiteCredito) AS nivel
FROM Clientes;

Bucles: WHILE, REPEAT y LOOP
-------------------------------------------------------------------------
MySQL ofrece tres estructuras de iteración. La diferencia clave está en cuándo se comprueba la condición.

WHILE … DO
Pre-prueba: puede ejecutarse 0 veces si la condición es falsa desde el inicio

REPEAT … UNTIL
Post-prueba: siempre se ejecuta al menos una vez

LOOP + LEAVE

Bucle infinito controlado manualmente con LEAVE al cumplirse la condición

-- WHILE: suma del 1 al 10
DECLARE i INT DEFAULT 1;
DECLARE suma INT DEFAULT 0;
WHILE i <= 10 DO
    SET suma = suma + i;
    SET i = i + 1;
END WHILE;

Procedimiento con parámetro
-------------------------------------------------------------------------
Crea un procedimiento GetEmpleadosDepartamento que reciba el nombre de un departamento como parámetro IN y devuelva los empleados que pertenecen a él ordenados por apellido.

DELIMITER $$
CREATE PROCEDURE GetEmpleadosDepartamento(
    IN pDepartamento VARCHAR(100))
BEGIN
    SELECT nombre, apellido, puesto
    FROM Empleados
    WHERE departamento = pDepartamento
    ORDER BY apellido;
END$$
DELIMITER ;

CALL GetEmpleadosDepartamento('Ventas');

Función de clasificación
-------------------------------------------------------------------------
Crea una función CalificacionAlumno que reciba una nota (0-10) y devuelva: 'Suspenso' (< 5), 'Aprobado' (5-6), 'Notable' (7-8) o 'Sobresaliente' (9-10).

DELIMITER $$
CREATE FUNCTION CalificacionAlumno(nota DECIMAL(4,2))
RETURNS VARCHAR(20)
BEGIN
    DECLARE calificacion VARCHAR(20);
    IF nota < 5 THEN
        SET calificacion = 'Suspenso';
    ELSEIF nota < 7 THEN
        SET calificacion = 'Aprobado';
    ELSEIF nota < 9 THEN
        SET calificacion = 'Notable';
    ELSE
        SET calificacion = 'Sobresaliente';
    END IF;
    RETURN calificacion;
END$$
DELIMITER ;

Procedimiento con bucle WHILE
-------------------------------------------------------------------------
Crea un procedimiento InsertarNumeros que, usando un bucle WHILE, inserte los números del 1 al N (parámetro IN) en una tabla numeros(valor INT).


DELIMITER $$
CREATE PROCEDURE InsertarNumeros(IN pN INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    WHILE i <= pN DO
        INSERT INTO numeros (valor) VALUES (i);
        SET i = i + 1;
    END WHILE;
END$$
DELIMITER ;
