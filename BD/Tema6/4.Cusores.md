Un cursor permite recorrer fila a fila el resultado de un SELECT dentro de un procedimiento almacenado. Siempre siguen el mismo ciclo de vida:

1. DECLARE
Declara el cursor asociado a una SELECT
2. OPEN
Inicializa el conjunto de resultados
3. FETCH
Recupera la siguiente fila en variables
4. CLOSE
Libera los recursos del cursor

El handler NOT FOUND es obligatorio: cuando FETCH llega al final, se activa esta condición. Sin él, el procedimiento lanzará un error.
-------------------------------------------------------------------------

Estructura completa con cursor
DELIMITER $$
CREATE FUNCTION listarEmails() RETURNS VARCHAR(4000)
BEGIN
    -- 1. Variables locales
    DECLARE terminado INT DEFAULT 0;
    DECLARE email VARCHAR(100) DEFAULT '';
    DECLARE lista  VARCHAR(4000) DEFAULT '';

    -- 2. Declarar cursor (siempre después de variables)
    DECLARE curEmail CURSOR FOR
        SELECT correo FROM Empleados;

    -- 3. Handler NOT FOUND (siempre al final de los DECLARE)
    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET terminado = 1;

    OPEN curEmail;
    bucle: LOOP
        FETCH curEmail INTO email;
        IF terminado = 1 THEN
            LEAVE bucle;
        END IF;
        SET lista = CONCAT(email, ';', lista);
    END LOOP bucle;
    CLOSE curEmail;

    RETURN lista;
END$$
DELIMITER ;

Construir listado con cursor
Crea un procedimiento ListarProductosBaratos que use un cursor para recorrer los productos con precio < 20 y construya una cadena con el formato "nombre1|nombre2|...", mostrándola al final con SELECT.

DELIMITER $$
CREATE PROCEDURE ListarProductosBaratos()
BEGIN
    DECLARE fin INT DEFAULT 0;
    DECLARE nomProd VARCHAR(200);
    DECLARE resultado VARCHAR(4000) DEFAULT '';

    DECLARE cur CURSOR FOR
        SELECT nombre FROM Productos
        WHERE precio < 20;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET fin = 1;

    OPEN cur;
    loop1: LOOP
        FETCH cur INTO nomProd;
        IF fin = 1 THEN LEAVE loop1; END IF;
        SET resultado = CONCAT(resultado,
            IF(resultado='','','|'), nomProd);
    END LOOP;
    CLOSE cur;

    SELECT resultado AS productos_baratos;
END$$
DELIMITER ;
